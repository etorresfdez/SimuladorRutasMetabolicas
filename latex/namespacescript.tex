\doxysubsubsection{Descripción detallada}
\begin{DoxyVerb}script de python para lanzar el solver de sistemas de ecuaciones diferenciales
para problemas de valores iniciales.

INPUT: Formato del archivo de entrada
    ecuaciones;- p[0]*x[0] + p[1]*x[1];- p[37]*x[5]*x[7] + p[38]*x[6]**2
    X0;1;0.5;2
    p;0.01;0.5;10
    tini;0
    tfin;100
    ciclos;1000
    method;Radau

OUTPUT: Formato del archivo de salida
    0.0;1.55307;0.573074
    1.0;1.6;0.03
    2.0;1.6;0.006
\end{DoxyVerb}
